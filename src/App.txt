// Use the deployed GitHub Pages address for Supabase email
confirmation. async function auth( email: string, password: string,
signup: boolean, name: string, role: string ) { if (!supabase) {
setNotice(‘Connect Supabase first using the setup guide.’); return; }

    const result = signup
      ? await supabase.auth.signUp({
          email,
          password,
          options: {
            emailRedirectTo: redirectUrl,
            data: {
              full_name: name,
              account_type: role
            }
          }
        })
      : await supabase.auth.signInWithPassword({
          email,
          password
        });

    if (result.error) {
      setNotice(result.error.message);
      return;
    }

    setUser(result.data.session?.user ?? null);
    setModal(null);

    if (signup) {
      setNotice(
        result.data.session
          ? 'Account created successfully.'
          : 'Account created. Please check your email and confirm your account.'
      );
    } else {
      setNotice('Welcome back.');
    }

}

async function addProduct(form: FormData) { if (!supabase || !user) {
setNotice(‘Sign in and connect Supabase to publish products.’); return;
}

    const { error } = await supabase.from('products').insert({
      seller_id: user.id,
      title: form.get('title'),
      category: form.get('category'),
      variety: form.get('variety') || null,
      price: Number(form.get('price')),
      unit: form.get('unit'),
      quantity: Number(form.get('quantity')),
      county: form.get('county'),
      description: form.get('description'),
      image_url: form.get('image_url') || null,
      status: 'active'
    });

    if (error) {
      setNotice(error.message);
      return;
    }

    setModal(null);
    setNotice('Your listing has been submitted.');

    const { data } = await supabase
      .from('products')
      .select('*,profiles(full_name,farm_name,county,rating)')
      .eq('status', 'active')
      .order('created_at', { ascending: false });

    if (data) setProducts(data as Product[]);

}

async function order(p: Product) { if (!supabase || !user) {
setCart([…cart, p]); setModal(‘login’); setNotice(‘Sign in to place an
order.’); return; }

    const { data, error } = await supabase.rpc('create_marketplace_order', {
      p_product_id: p.id,
      p_quantity: 1
    });

    if (error) {
      setNotice(error.message);
      return;
    }

    setNotice(
      `Order ${String(data).slice(0, 8)} created. Continue to secure checkout from your dashboard when payment is enabled.`
    );

}
